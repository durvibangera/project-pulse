## 0. Project Overview

**Goal:** Build an end-to-end Clinical Intelligence Platform that demonstrates Real-World Evidence (RWE) analysis, multi-vertical predictive modeling, ML model monitoring, and NLP-powered research summarization — directly mapping to the Abbott Global Data Science & Analytics (GDSA) role.

**Datasets:**
| ID | Dataset | Source | Role in Project |
|---|---|---|---|
| D1 | Diabetes 130-US Hospitals | UCI | Primary ML training (Metabolic Vertical) |
| D2 | NHANES (2015-16 + 2021-23) | CDC | Drift simulation (Model Monitoring) |
| D3 | Chronic Disease Prediction | Kaggle | Multi-vertical feature enrichment |
| D4 | Statlog Heart Disease | UCI | Cardiovascular Vertical ML |
| D5 | Drug Reviews (Druglib.com) | UCI | NLP / RAG Summarization Engine |

**Four Layers:**
- **Layer 1 — Feature Store:** Unified data warehouse across all 5 datasets
- **Layer 2 — ML Engine:** Diabetes + Cardiovascular predictive models
- **Layer 3 — Sentinel Suite:** NHANES drift detection (2015-16 → 2021-23)
- **Layer 4 — Evidence Summarizer:** Drug review NLP + RAG pipeline

## 3. Layer 1 — Unified Feature Store (Data Engineering)

### 3.1 Shared Schema Design

The core insight: **Diabetes 130, NHANES, Statlog, and Chronic Disease all share a common biomarker vocabulary.** Map them to a unified schema before modeling.

**Master Schema (target columns):**

```python
MASTER_SCHEMA = {
    # Demographics
    "patient_id": str,
    "age": float,          # years (standardized)
    "gender": int,         # 0=F, 1=M
    "bmi": float,

    # Metabolic Panel
    "glucose": float,      # mg/dL
    "hba1c": float,        # % (if available)
    "insulin": float,

    # Cardiovascular Panel
    "blood_pressure_sys": float,   # systolic mmHg
    "blood_pressure_dia": float,   # diastolic mmHg
    "cholesterol": float,
    "heart_rate": float,

    # Renal / Other
    "serum_creatinine": float,
    "skin_thickness": float,

    # Labels
    "diabetes_label": int,    # 0/1
    "cardio_label": int,      # 0/1
    "source_dataset": str,    # "diabetes_130" | "nhanes" | "statlog" | "chronic"
    "nhanes_cycle": str,      # "2015_16" | "2021_23" | None
```

### 3.2 Dataset Loaders (`src/data/loader.py`)

```python
import pandas as pd
import pyreadstat
import os

def load_diabetes_130(path="data/raw/diabetes_130/diabetic_data.csv"):
    df = pd.read_csv(path)
    df = df.replace("?", pd.NA)
    return df

def load_nhanes_cycle(cycle_dir: str):
    """
    Loads and merges Demographics (DEMO), Lab (LAB/GHB), and
    Blood Pressure (BPX) XPT files from a single NHANES cycle.
    Joins on SEQN (Respondent Sequence Number).
    """
    xpt_files = {
        "demo": None,
        "lab": None,
        "bpx": None,
        "ghb": None,   # HbA1c
    }
    for fname in os.listdir(cycle_dir):
        lower = fname.lower()
        if "demo" in lower:   xpt_files["demo"] = fname
        elif "ghb" in lower:  xpt_files["ghb"] = fname
        elif "bpx" in lower:  xpt_files["bpx"] = fname
        elif lower.endswith(".xpt"): xpt_files["lab"] = fname

    frames = {}
    for key, fname in xpt_files.items():
        if fname:
            fpath = os.path.join(cycle_dir, fname)
            df, _ = pyreadstat.read_xport(fpath)
            frames[key] = df

    # Merge all on SEQN
    merged = frames["demo"]
    for key in ["lab", "bpx", "ghb"]:
        if key in frames:
            merged = merged.merge(frames[key], on="SEQN", how="left")
    return merged

def load_statlog(path="data/raw/statlog_heart/heart.dat"):
    cols = [
        "age", "sex", "chest_pain_type", "blood_pressure_sys",
        "cholesterol", "fasting_blood_sugar", "resting_ecg",
        "max_heart_rate", "exercise_angina", "st_depression",
        "st_slope", "major_vessels", "thal", "cardio_label"
    ]
    df = pd.read_csv(path, sep=" ", header=None, names=cols)
    df["cardio_label"] = (df["cardio_label"] == 2).astype(int)  # 2=disease
    return df

def load_chronic_disease(path="data/raw/chronic_disease/chronic_disease.csv"):
    return pd.read_csv(path)

def load_drug_reviews(path="data/raw/drug_reviews/druglib_reviews.csv"):
    df = pd.read_csv(path, sep="\t", encoding="latin-1")
    return df
```

### 3.3 Cleaning & Standardization (`src/data/cleaner.py`)

```python
import pandas as pd
import numpy as np

def clean_diabetes_130(df: pd.DataFrame) -> pd.DataFrame:
    """Map Diabetes 130 to master schema."""
    out = pd.DataFrame()
    out["patient_id"] = df["encounter_id"].astype(str)
    out["gender"] = df["gender"].map({"Male": 1, "Female": 0})
    # Age is a range string like "[70-80)" — extract midpoint
    out["age"] = df["age"].str.extract(r"\[(\d+)")[0].astype(float) + 5
    out["hba1c"] = pd.to_numeric(df["A1Cresult"].map(
        {">8": 9.0, ">7": 7.5, "Norm": 5.5, "None": np.nan}
    ), errors="coerce")
    out["diabetes_label"] = (df["readmitted"] != "NO").astype(int)
    out["cardio_label"] = np.nan
    out["source_dataset"] = "diabetes_130"
    out["nhanes_cycle"] = None
    return out

def clean_nhanes(df: pd.DataFrame, cycle: str) -> pd.DataFrame:
    """Map NHANES XPT merged frame to master schema."""
    out = pd.DataFrame()
    out["patient_id"] = df["SEQN"].astype(str) + f"_{cycle}"
    out["age"] = df.get("RIDAGEYR")
    out["gender"] = df.get("RIAGENDR").map({1: 1, 2: 0})  # 1=Male, 2=Female
    out["bmi"] = df.get("BMXBMI")
    out["glucose"] = df.get("LBXGLU")        # Fasting glucose mg/dL
    out["hba1c"] = df.get("LBXGH")           # HbA1c %
    out["blood_pressure_sys"] = df.get("BPXSY1")
    out["blood_pressure_dia"] = df.get("BPXDI1")
    # Diabetes label: HbA1c >= 6.5 OR glucose >= 126
    out["diabetes_label"] = (
        (out["hba1c"] >= 6.5) | (out["glucose"] >= 126)
    ).astype(float)
    out["cardio_label"] = np.nan
    out["source_dataset"] = "nhanes"
    out["nhanes_cycle"] = cycle
    return out

def clean_statlog(df: pd.DataFrame) -> pd.DataFrame:
    out = df.copy()
    out["patient_id"] = "statlog_" + out.index.astype(str)
    out["gender"] = out["sex"]
    out["diabetes_label"] = np.nan
    out["source_dataset"] = "statlog"
    out["nhanes_cycle"] = None
    return out[["patient_id", "age", "gender", "blood_pressure_sys",
                "cholesterol", "max_heart_rate", "cardio_label",
                "diabetes_label", "source_dataset", "nhanes_cycle"]]

def handle_missing(df: pd.DataFrame) -> pd.DataFrame:
    """Fill numeric NaNs with median (clinical safe default)."""
    num_cols = df.select_dtypes(include=[np.number]).columns
    df[num_cols] = df[num_cols].fillna(df[num_cols].median())
    return df

def remove_outliers_iqr(df: pd.DataFrame, col: str, factor: float = 3.0):
    """Winsorize extreme values using IQR."""
    Q1, Q3 = df[col].quantile(0.25), df[col].quantile(0.75)
    IQR = Q3 - Q1
    df[col] = df[col].clip(Q1 - factor * IQR, Q3 + factor * IQR)
    return df
```

### 3.4 Building the Master Patient Table (`src/data/feature_store.py`)

```python
import pandas as pd
from loader import (load_diabetes_130, load_nhanes_cycle, load_statlog)
from cleaner import (clean_diabetes_130, clean_nhanes, clean_statlog, handle_missing)

MASTER_COLS = [
    "patient_id", "age", "gender", "bmi", "glucose", "hba1c",
    "blood_pressure_sys", "blood_pressure_dia", "cholesterol",
    "heart_rate", "diabetes_label", "cardio_label",
    "source_dataset", "nhanes_cycle"
]

def build_feature_store():
    frames = []

    # D1 — Diabetes 130
    frames.append(clean_diabetes_130(load_diabetes_130()))

    # D2a — NHANES 2015-16 (training baseline)
    nhanes_2015 = clean_nhanes(
        load_nhanes_cycle("data/raw/nhanes/2015_16"), cycle="2015_16"
    )
    nhanes_2015.to_parquet("data/processed/nhanes_2015.parquet", index=False)
    frames.append(nhanes_2015)

    # D2b — NHANES 2021-23 (production drift stream)
    nhanes_2021 = clean_nhanes(
        load_nhanes_cycle("data/raw/nhanes/2021_23"), cycle="2021_23"
    )
    nhanes_2021.to_parquet("data/processed/nhanes_2021.parquet", index=False)
    frames.append(nhanes_2021)

    # D4 — Statlog
    frames.append(clean_statlog(load_statlog()))

    # Combine into Master Patient Table
    master = pd.concat(frames, ignore_index=True)
    master = master.reindex(columns=MASTER_COLS)
    master = handle_missing(master)
    master.to_parquet("data/processed/master_patient_table.parquet", index=False)

    print(f"[PULSE] Master Patient Table built: {master.shape}")
    return master

if __name__ == "__main__":
    build_feature_store()
```

## 4. Layer 2 — Multi-Vertical ML Engine

### 4.1 Shared Base Model (`src/models/base_model.py`)

```python
import mlflow
import mlflow.sklearn
import shap
import pandas as pd
from sklearn.model_selection import StratifiedKFold, cross_val_score
from sklearn.metrics import roc_auc_score, f1_score, classification_report
from xgboost import XGBClassifier

class BaseModel:
    def __init__(self, name: str, features: list, label: str):
        self.name = name
        self.features = features
        self.label = label
        self.model = XGBClassifier(
            n_estimators=300,
            max_depth=6,
            learning_rate=0.05,
            subsample=0.8,
            colsample_bytree=0.8,
            eval_metric="logloss",
            random_state=42
        )

    def train(self, X_train, y_train, experiment_name="PULSE"):
        mlflow.set_experiment(experiment_name)
        with mlflow.start_run(run_name=self.name):
            self.model.fit(X_train, y_train)
            cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
            auc_scores = cross_val_score(
                self.model, X_train, y_train, cv=cv, scoring="roc_auc"
            )
            mlflow.log_metric("cv_auc_mean", auc_scores.mean())
            mlflow.log_metric("cv_auc_std", auc_scores.std())
            mlflow.log_params(self.model.get_params())
            mlflow.sklearn.log_model(self.model, artifact_path=self.name)
            print(f"[{self.name}] CV AUC: {auc_scores.mean():.4f} ± {auc_scores.std():.4f}")

    def evaluate(self, X_test, y_test):
        preds = self.model.predict(X_test)
        proba = self.model.predict_proba(X_test)[:, 1]
        print(f"\n=== {self.name} Evaluation ===")
        print(classification_report(y_test, preds))
        print(f"ROC-AUC: {roc_auc_score(y_test, proba):.4f}")
        return {"auc": roc_auc_score(y_test, proba), "f1": f1_score(y_test, preds)}

    def explain(self, X_sample):
        """SHAP feature importance — call after train()."""
        explainer = shap.TreeExplainer(self.model)
        shap_values = explainer.shap_values(X_sample)
        shap.summary_plot(shap_values, X_sample, feature_names=self.features)
        return shap_values
```

### 4.2 Metabolic Vertical — Diabetes Model (`src/models/diabetes_model.py`)

```python
import pandas as pd
import joblib
from sklearn.model_selection import train_test_split
from base_model import BaseModel

DIABETES_FEATURES = [
    "age", "gender", "bmi", "glucose", "hba1c",
    "blood_pressure_sys", "blood_pressure_dia"
]

def train_diabetes_model():
    master = pd.read_parquet("data/processed/master_patient_table.parquet")
    df = master[
        master["diabetes_label"].notna() &
        master["source_dataset"].isin(["diabetes_130", "nhanes"])
    ].copy()

    X = df[DIABETES_FEATURES].fillna(df[DIABETES_FEATURES].median())
    y = df["diabetes_label"].astype(int)

    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=0.2, stratify=y, random_state=42
    )

    model = BaseModel("DiabetesClassifier", DIABETES_FEATURES, "diabetes_label")
    model.train(X_train, y_train)
    metrics = model.evaluate(X_test, y_test)
    model.explain(X_test.sample(200, random_state=42))

    joblib.dump(model.model, "models/diabetes_xgb.pkl")
    joblib.dump(DIABETES_FEATURES, "models/diabetes_features.pkl")
    return model, metrics

if __name__ == "__main__":
    train_diabetes_model()
```

### 5.2 Drift Detector (`src/monitoring/drift_detector.py`)

```python
import pandas as pd
import joblib
from evidently.report import Report
from evidently.metric_preset import DataDriftPreset, ClassificationPreset
from evidently.metrics import DatasetDriftMetric, DataDriftTable, ColumnDriftMetric

DRIFT_FEATURES = [
    "age", "gender", "bmi", "glucose", "hba1c",
    "blood_pressure_sys", "blood_pressure_dia"
]

def run_drift_report(
    reference_path="data/processed/nhanes_2015.parquet",
    current_path="data/processed/nhanes_2021.parquet",
    model_path="models/diabetes_xgb.pkl",
    output_path="reports/model_health_report.html"
):
    ref = pd.read_parquet(reference_path)[DRIFT_FEATURES + ["diabetes_label"]].dropna()
    cur = pd.read_parquet(current_path)[DRIFT_FEATURES + ["diabetes_label"]].dropna()

    model = joblib.load(model_path)
    ref["prediction"] = model.predict(ref[DRIFT_FEATURES])
    cur["prediction"] = model.predict(cur[DRIFT_FEATURES])

    report = Report(metrics=[
        DatasetDriftMetric(),
        DataDriftTable(),
        ColumnDriftMetric(column_name="glucose"),
        ColumnDriftMetric(column_name="hba1c"),
        ColumnDriftMetric(column_name="bmi"),
        ClassificationPreset(),
    ])

    report.run(reference_data=ref, current_data=cur)
    report.save_html(output_path)
    print(f"[SENTINEL] Health Report saved → {output_path}")

    result = report.as_dict()
    drift_detected = result["metrics"][0]["result"]["dataset_drift"]

    if drift_detected:
        print("[ALERT] Significant data drift detected! Retraining recommended.")
    else:
        print("[OK] No significant drift detected.")

    return result

if __name__ == "__main__":
    run_drift_report()
```

### 6.2 Index Drug Reviews (`src/nlp/rag_pipeline.py`)

```python
import pandas as pd
from langchain_community.vectorstores import Chroma
from langchain_community.embeddings import HuggingFaceEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.docstore.document import Document

EMBED_MODEL = "sentence-transformers/all-MiniLM-L6-v2"   # Free, local
CHROMA_DIR = "./chroma_db"

def build_drug_index(reviews_path="data/raw/drug_reviews/druglib_reviews.csv"):
    df = pd.read_csv(reviews_path, sep="\t", encoding="latin-1")

    df["text"] = (
        "Drug: " + df["urlDrugName"].fillna("Unknown") + "\n"
        + "Condition: " + df["condition"].fillna("Unknown") + "\n"
        + "Benefits: " + df["benefitsReview"].fillna("") + "\n"
        + "Side Effects: " + df["sideEffectsReview"].fillna("") + "\n"
        + "Comments: " + df["commentsReview"].fillna("") + "\n"
        + "Rating: " + df["rating"].astype(str)
    )

    docs = [
        Document(
            page_content=row["text"],
            metadata={
                "drug": row.get("urlDrugName", ""),
                "condition": row.get("condition", ""),
                "rating": row.get("rating", 0),
            }
        )
        for _, row in df.iterrows()
    ]

    splitter = RecursiveCharacterTextSplitter(chunk_size=500, chunk_overlap=50)
    chunks = splitter.split_documents(docs)

    embeddings = HuggingFaceEmbeddings(model_name=EMBED_MODEL)
    vectorstore = Chroma.from_documents(
        documents=chunks,
        embedding=embeddings,
        persist_directory=CHROMA_DIR
    )
    vectorstore.persist()
    print(f"[NLP] Indexed {len(chunks)} chunks into ChromaDB.")
    return vectorstore

def load_drug_index():
    embeddings = HuggingFaceEmbeddings(model_name=EMBED_MODEL)
    return Chroma(persist_directory=CHROMA_DIR, embedding_function=embeddings)
```

### 6.3 Summarizer Chain (`src/nlp/summarizer.py`)

```python
from langchain.chains import RetrievalQA
from langchain_community.chat_models import ChatOpenAI
from rag_pipeline import load_drug_index

SYSTEM_PROMPT = """
You are a clinical evidence summarizer for a Global Data Science team.
Given patient drug reviews, produce a concise structured summary with:
1. Drug effectiveness (1-2 sentences)
2. Most common side effects
3. Overall patient sentiment (Positive / Mixed / Negative)
4. Key recommendation for the clinical team
Keep your response factual, neutral, and evidence-based.
"""

def summarize_drug(condition: str, drug: str = None, top_k: int = 10):
    vectorstore = load_drug_index()
    retriever = vectorstore.as_retriever(
        search_kwargs={
            "k": top_k,
            "filter": {"condition": condition} if condition else None
        }
    )

    # Option A: OpenAI (paid, ~$0.01/query)
    llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)

    # Option B: Ollama / Llama 3 (free, local) — uncomment to use
    # from langchain_community.chat_models import ChatOllama
    # llm = ChatOllama(model="llama3")

    qa_chain = RetrievalQA.from_chain_type(
        llm=llm,
        retriever=retriever,
        chain_type="stuff",
    )

    query = f"Summarize patient experiences with {drug or 'medications'} for {condition}."
    return qa_chain.run(query)
```

## 9. Key Technical Decisions

1. **XGBoost over Logistic Regression** — clinical biomarkers have non-linear interactions (e.g., glucose × HbA1c × BMI). XGBoost handles this naturally; SHAP makes it interpretable for clinical audiences.

2. **Evidently AI over custom drift code** — production-grade library, generates shareable HTML reports, mirrors real-world MLOps tooling used in pharma and medtech.

3. **ChromaDB + HuggingFace Embeddings** — no API cost for indexing. Embeddings run fully locally. Only the final summarization step calls an LLM (GPT-4o-mini ~$0.01/query, or free via Ollama).

4. **NHANES XPT files** — must use `pyreadstat`, not pandas directly. Always join on `SEQN`. File naming conventions differ by cycle — the loader uses keyword matching to handle this automatically.

5. **Parquet over CSV** — for processed data, Parquet is 5–10x faster at read time and preserves column dtypes. Critical for the ~100K+ row master table used across multiple scripts.

6. **Stratified splits** — clinical datasets are often class-imbalanced (e.g., diabetic vs non-diabetic). Always use `stratify=y` in `train_test_split` and `StratifiedKFold` for CV.
